006: /// Turns a completed [Walk] into a [MemoryCard] - a diary line written in the
007: /// companion's voice. Fully deterministic given a walk + seed, so it is unit
008: /// testable and never produces unsafe AI text (curated templates only).
009: class MemoryGenerator {
010:   const MemoryGenerator();
011: 
012:   /// Generate a memory for [walk]. [seed] varies the wording; [dayIndex] labels it.
013:   static MemoryCard generate(
014:     Walk walk, {
015:     required int seed,
016:     required int dayIndex,
017:     bool premium = false,
018:   }) {
019:     final lead = walk.lead;
020:     final leadPlace = walk.stops.isNotEmpty ? walk.stops.first : null;
021:     final lastPlace = walk.stops.isNotEmpty ? walk.stops.last : leadPlace;
022:     final firstLabel = leadPlace?.displayLabel ?? PlaceType.park.label;
023:     final lastLabel = lastPlace?.displayLabel ?? firstLabel;
024: 
025:     final templates = _diaryTemplates[lead]!;
026:     final base = templates[seed.abs() % templates.length];
027:     var diary = base
028:         .replaceAll('{first}', firstLabel)
029:         .replaceAll('{last}', lastLabel)
030:         .replaceAll('{time}', walk.time.label)
031:         .replaceAll('{weather}', walk.weather.label);
032: 
033:     if (premium) {
034:       diary = '$diary ${_voiceFlourish[lead]!}';
035:     }
036: 
037:     final titles = _titles[lead]!;
038:     final title = titles[seed.abs() % titles.length];
039: 
040:     return MemoryCard(
041:       id: 'mem-$dayIndex-${seed.abs()}',
042:       title: title,
043:       diary: diary,
044:       lead: lead,
045:       places: walk.stops.map((s) => s.place).toList(),
046:       placeHints: walk.stops.map((s) => s.displayLabel).toList(),
047:       time: walk.time,
048:       weather: walk.weather,
049:       dayLabel: '$dayIndex日目',
050:       premium: premium,
051:     );
052:   }
053: 
054:   static const Map<DnaTrait, List<String>> _titles = {
055:     DnaTrait.vitality: ['明るく早いひと回り', '動き出す道', '朝の勢い'],
056:     DnaTrait.calm: ['ゆっくりした時間', '水辺の静けさ', '急がなくていい日'],
057:     DnaTrait.curiosity: ['今日は新しい発見', '知らない角を曲がって', '目が覚めるさんぽ'],
058:     DnaTrait.warmth: ['やさしい一日', '家の近くで', '金色で穏やか'],
059:     DnaTrait.focus: ['整った道すじ', 'まっすぐな時間', '頭が片づく日'],
060:     DnaTrait.wonder: ['夕暮れの小さな魔法', '明かりと静けさ', '夢みたいな寄り道'],
