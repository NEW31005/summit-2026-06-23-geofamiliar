007: /// One stop on a walk - a single place visited under the day's time/weather.
008: class RouteStop {
009:   const RouteStop(this.place, {this.context});
010: 
011:   final PlaceType place;
012:   final PlaceContext? context;
013: 
014:   PlaceContext get effectiveContext => context ?? PlaceContext.manual(place);
015:   String get displayLabel => effectiveContext.displayHint;
016: 
017:   Map<String, dynamic> toJson() => {
018:     'place': place.name,
019:     if (context != null) 'context': context!.toJson(),
020:   };
021: 
022:   factory RouteStop.fromJson(Map<String, dynamic> json) {
023:     final place = PlaceType.fromName(json['place'] as String);
024:     final contextJson = json['context'];
025:     final context = contextJson is Map<String, dynamic>
026:         ? PlaceContext.fromJson(contextJson)
027:         : null;
028:     return RouteStop(place, context: context);
029:   }
030: }
